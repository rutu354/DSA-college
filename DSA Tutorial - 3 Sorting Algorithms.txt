#include <stdio.h>

void swap(int *a, int *b)
{
   int temp;
   temp = *a;
   *a = *b;
   *b = temp;
}

void bs(int arr[], int n)
{
   int i, j;
   int swapped;
   for(i = 0; i < n - 1; i++)
   {
      swapped = 0;
      for(j = 0; j < n - i - 1; j++)
      {
          if(arr[j] > arr[j + 1])
          {
              swap(&arr[j], &arr[j + 1]);
              swapped = 1;
          }
      }
      if(swapped == 0)
      {
          break;
      }
   }
   printf("Sorted array using Bubble Sort: ");
   for(i = 0; i < n; i++)
   {
      printf("%d ", arr[i]);
   }
   printf("\n");
}

void ss(int arr[], int n)
{
  int i, j;
  int minidx;
  for(i = 0; i < n - 1; i++)
  {
     minidx = i;
     for(j = i + 1; j < n; j++)
     {
         if(arr[j] < arr[minidx])
         {
             minidx = j;
         }
     }
     if(minidx != i)
     {
         swap(&arr[i], &arr[minidx]);
     }
  }
  printf("Sorted array using Selection Sort: ");
  for(i = 0; i < n; i++)
  {
     printf("%d ", arr[i]);
  }
  printf("\n");
}

void is(int arr[], int n)
{
  int i, j;
  int key;
  for(i = 1; i < n; i++)
  {
     key = arr[i];
     j = i - 1;
     while(j >= 0 && arr[j] > key)
     {
         arr[j + 1] = arr[j];
         j--;
     }
     arr[j + 1] = key;
  }
  printf("Sorted array using Insertion Sort: ");
  for(i = 0; i < n; i++)
  {
     printf("%d ", arr[i]);
  }
  printf("\n");
}

int main()
{
   int n;
   int choice;
   int i;
   printf("Enter number of elements: ");
   scanf("%d", &n);
   int arr[n];
   printf("Enter %d elements:\n", n);
   for(i = 0; i < n; i++)
   {
      scanf("%d", &arr[i]);
   }
   printf("\n1. Bubble Sort\n");
   printf("2. Selection Sort\n");
   printf("3. Insertion Sort\n");
   printf("Enter your choice: ");
   scanf("%d", &choice);
   switch(choice)
   {
      case 1:
          bs(arr, n);
          break;
      case 2:
          ss(arr, n);
          break;
      case 3:
          is(arr, n);
          break;
      default:
          printf("Invalid choice\n");
   }
   return 0;
}
